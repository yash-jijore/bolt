#fractional knapsack


def fractional_knapsack(arr, W):

    # Sort by value/weight ratio in descending order
    arr.sort(key=lambda x: x[0] / x[1], reverse=True)

    curr_weight = 0
    final_value = 0

    for i in range(len(arr)):

        # If complete item can fit
        if curr_weight + arr[i][1] <= W:
            curr_weight += arr[i][1]
            final_value += arr[i][0]

        # If complete item cannot fit
        else:
            remain = W - curr_weight

            cost = (arr[i][0] / arr[i][1]) * remain

            final_value += cost
            break

    return final_value


# value, weight
arr = [
    (200, 50),
    (60, 10),
    (100, 50),
    (100, 20)
]

W = 80

print("Maximum value:", fractional_knapsack(arr, W))
