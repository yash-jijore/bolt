
# FIBONACCI

def recursive(n):
    if n <= 1:
        return n
    return recursive(n-1) + recursive(n-2)


def recursive_series(n):
    for i in range(n):
        print(recursive(i), end=" ")
    print()


def non_recursive(n):
    a, b = 0, 1

    for i in range(n):
        print(a, end=" ")
        c = a + b
        a = b
        b = c


n = 4

if n > 0:
    recursive_series(n)
    non_recursive(n)
